graph TD

    %% Define Actors
    Guest((Guest/Visitor))
    User((Authenticated User))
    Admin((Administrator))
    AI[[AI Vision Service]]

    %% Public Access
    subgraph "Public Entry"
        Landing[Landing Page / Hero Section]
        Register[Sign Up Page]
        Login[Login / Google OAuth]
    end

    %% User Dashboard Logic
    subgraph "User Experience (Dashboard)"
        Scanner[AICarScanner: Upload Photos]
        JobStatus[Real-time Processing Status]
        ViewEst[View Damage Analysis]
        History[View Past Estimates]
        Profile[Manage Profile & Avatar]
        PDF[Export PDF Report]
    end

    %% Admin Logic
    subgraph "Administration"
        Stats[Platform Usage Stats]
        ManageUsers[Ban/Unban/Delete Users]
        Messages[Manage Support Conversations]
    end

    %% Database & Systems
    subgraph "Backend Infrastructure"
        DB[(Prisma / Database)]
        Email[Email Service: Verification/Reset]
    end

    %% Relationships
    Guest --> Landing
    Landing --> Register
    Landing --> Login

    Login --> User
    User --> Scanner
    User --> History
    User --> Profile
    
    Scanner -.-> AI
    AI -.-> JobStatus
    JobStatus --> ViewEst
    ViewEst --> PDF

    User --> Admin
    Admin --> Stats
    Admin --> ManageUsers
    Admin --> Messages

    %% Data Connections
    Scanner & Profile & ManageUsers --> DB
    Register & Login --> Email
